cmake_minimum_required(VERSION 3.18)
project(i_crossplane_core LANGUAGES CXX)
add_library(im_crossplane STATIC src/im_crossplane.cpp)
target_include_directories(im_crossplane PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(im_crossplane PUBLIC cxx_std_17)
if(NOT MSVC)
  target_compile_options(im_crossplane PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(im_crossplane_test tests/test_crossplane.cpp)
target_link_libraries(im_crossplane_test PRIVATE im_crossplane)
enable_testing()
add_test(NAME im_crossplane_20k COMMAND im_crossplane_test)
